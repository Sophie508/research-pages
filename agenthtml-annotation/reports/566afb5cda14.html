View core computation 
SELECT name, idx, COUNT(*) AS trading_days,
       ROUND(AVG(100 * (hi - lo) / op), 4)
         AS avg_intraday_volatility_pct
FROM parsed JOIN asia USING (idx)
WHERE dt >= DATE '2020-01-01'
  AND op IS NOT NULL AND hi IS NOT NULL AND lo IS NOT NULL
  AND op != 0
GROUP BY name, idx
ORDER BY avg_intraday_volatility_pct DESC;
