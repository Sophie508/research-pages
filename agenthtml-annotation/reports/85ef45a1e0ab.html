Method

  • 2
    Identify Asia indexes. The selected tickers were HSI, 000001.SS, N225, 399001.SZ, NSEI, and TWII.

    Filter dates. Mixed date strings were parsed with three formats, then filtered to observations on or after 2020-01-01.

Core Query Shape

WITH parsed AS (
  SELECT "Index" AS idx,
    COALESCE(
      try_strptime("Date", '%Y-%m-%d %H:%M:%S'),
      try_strptime("Date", '%d %b %Y, %H:%M'),
      try_strptime("Date", '%B %d, %Y at %I:%M %p')
    ) AS dt,
    "Open" AS op, "High" AS hi, "Low" AS lo
  FROM index_trade
), asia(idx, name) AS (
  VALUES
    ('HSI','Hang Seng Index'),
    ('000001.SS','Shanghai Composite'),
    ('N225','Nikkei 225'),
    ('399001.SZ','Shenzhen Component'),
    ('NSEI','Nifty 50'),
    ('TWII','Taiwan Weighted')
)
SELECT name, idx, COUNT(*) AS trading_days,
  ROUND(AVG(100 * (hi - lo) / op), 4)
FROM parsed JOIN asia USING(idx)
WHERE dt >= DATE '2020-01-01' AND op > 0
GROUP BY name, idx
ORDER BY 4 DESC;
