import duckdb

c = duckdb.connect("data/indextrade_query.db", read_only=True)

q = """
WITH parsed AS (
  SELECT "Index" AS idx,
    COALESCE(
      try_strptime("Date", '%Y-%m-%d %H:%M:%S'),
      try_strptime("Date", '%d %b %Y, %H:%M'),
      try_strptime("Date", '%B %d, %Y at %I:%M %p')
    ) AS dt,
    "Open" AS op, "High" AS hi, "Low" AS lo
  FROM index_trade
), asia(idx, name) AS (
  VALUES
    ('HSI',       'Hang Seng Index'),
    ('000001.SS', 'Shanghai Composite'),
    ('N225',      'Nikkei 225'),
    ('399001.SZ', 'Shenzhen Component'),
    ('NSEI',      'Nifty 50'),
    ('TWII',      'Taiwan Weighted')
)
SELECT
  name, idx,
  COUNT(*)                            AS trading_days,
  ROUND(AVG((hi - lo) / op * 100), 4) AS avg_volatility_pct
FROM parsed
JOIN asia USING (idx)
WHERE dt >= '2020-01-01'
  AND op > 0
GROUP BY name, idx
ORDER BY avg_volatility_pct DESC
"""

for row in c.execute(q).fetchall():
    print(f"{row[0]:20s} | {row[1]:12s} | {row[2]:3d} | {row[3]:.4f}")
